Quick start

Basic Jelly.js usage. 

const jelly = new Jelly({
  preset: "petBlob",
  color: 0xe00072,
  eyes: true,
  elasticity: 2
});

scene.add(jelly.mesh);

function animate() {
  jelly.update(clock.getDelta());
  renderer.render(scene, camera);
}

animate();
